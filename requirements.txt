Django==5.2.17

# Выгрузка расписания в xlsx. Чтение файлов больше не нужно: импорт из
# Excel/CSV удалён, сотрудницы вносят расписание через админку.
openpyxl==3.1.5

# Рабочий сервер для прода. На Windows не запустится (только Linux и macOS),
# но ставится без ошибок — нужен, чтобы `pip install -r requirements.txt` на
# сервере поднял всё сразу. Локальная разработка идёт через runserver.
#
# ВНИМАНИЕ: на Windows и macOS этот пакет установить можно, а запустить нельзя.
# Проверялся на Python 3.14.5 только на установку; запуск на сервере проверить
# нужно там же. Если при старте будет ошибка про версию Python, установи
# сервер на 3.12 или 3.13.
gunicorn==26.2.0

# Рабочий сервер для Windows. Локальный показ коллегам идёт через него:
# gunicorn на Windows не запускается, а runserver однопоточный — при
# параллельной загрузке страницы начальник видит «страница не отвечает».
# waitress работает на всех трёх платформах, поэтому на сервере можно
# оставить тот же способ запуска, что и локально.
waitress==3.0.2
